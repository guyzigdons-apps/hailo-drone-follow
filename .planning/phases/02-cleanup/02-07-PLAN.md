  class SharedUIState:
      def __init__(self):
          self._lock = threading.Lock()
          self._cond = threading.Condition(self._lock)   # shares the same underlying lock
          self._frame_seq: int = 0                       # monotonic counter
          self._frame_jpeg: Optional[bytes] = None
          self._frame_snapshot: Optional[dict] = None
          # ... rest unchanged (detections, velocity, perf, etc.)

      def update_frame(self, jpeg_bytes: bytes) -> None:
          """Producer. Each frame increments _frame_seq atomically."""
          with self._cond:
              self._frame_jpeg = jpeg_bytes
              self._frame_snapshot = {...}  # whatever the current snapshot is
              self._frame_seq += 1
              self._cond.notify_all()       # SAFE — under the lock

      def wait_frame(self, last_seen: int, timeout: float = 1.0) -> "tuple[Optional[bytes], int]":
          """Returns (jpeg, new_seq) when frame_seq > last_seen, or (None, last_seen) on timeout."""
          with self._cond:
              ok = self._cond.wait_for(lambda: self._frame_seq > last_seen, timeout=timeout)
              if not ok:
                  return None, last_seen
              return self._frame_jpeg, self._frame_seq

      def wait_frame_with_detections(self, last_seen: int, timeout: float = 1.0):
          with self._cond:
              ok = self._cond.wait_for(lambda: self._frame_seq > last_seen, timeout=timeout)
              if not ok:
                  return None, None, last_seen
              return self._frame_jpeg, self._frame_snapshot, self._frame_seq

  CRITICAL: `notify_all()` MUST be called INSIDE the `with self._cond:` block. Python docs are explicit; notifications outside the lock race with `wait_for`'s predicate re-check.

  LOCK SHARING: `Condition(self._lock)` uses the same underlying lock. Other `with self._lock:` blocks (e.g. `update_detections`, `update_velocity`, `update_perf`) can stay as-is — they don't need to call `notify_all`. Only `update_frame` migrates to `with self._cond:`. Both names point at the same lock primitive.
