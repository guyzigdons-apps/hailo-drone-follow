```python
@classmethod
def from_json(cls, path: str) -> "ControllerConfig":
    with open(path) as f:
        data = json.load(f)
    valid_names = {field.name for field in fields(cls)}
    filtered = {k: v for k, v in data.items() if k in valid_names}
    return cls(**filtered)
```
