Pseudocode for run_robot_loop body (CONTEXT lines 68-105):
```python
async def run_robot_loop(robot, shared_state, config, shutdown):
    await robot.connect()
    await robot.start_session()
    last_seen_t = None
    last_cmd = RobotCommand()
    last_detection = None
    tick_dt = 1.0 / config.control_loop_hz
    try:
        while not shutdown.is_set():
            tick_start = time.monotonic()
            detection = shared_state.current_detection()
            if detection is not None:
                safety_ctx = SafetyContext.from_detection(detection)
                cmd = controller.compute(detection, robot.caps, config)
                await robot.send_command(cmd, safety_ctx)
                last_seen_t = time.monotonic()
                last_cmd = cmd
                last_detection = detection
            else:
                lost_for = (time.monotonic() - last_seen_t) if last_seen_t else math.inf
                if lost_for < config.search_enter_delay_s:
                    safety_ctx = (SafetyContext.from_detection(last_detection)
                                  if last_detection is not None
                                  else SafetyContext.lost())
                    await robot.send_command(last_cmd, safety_ctx)
                else:
                    await robot.on_target_lost(last_detection)
            elapsed = time.monotonic() - tick_start
            await asyncio.sleep(max(0.0, tick_dt - elapsed))
    finally:
        await robot.send_zero()
        await robot.shutdown()
```

         ```python
         def compute(
             detection: Detection,
             caps: Capabilities,
             config: ControllerConfig,
         ) -> RobotCommand:
             """Pure controller — emit a RobotCommand from a Detection.

             Writes only the channels in caps.axes:
             - forward_m_s: emitted only when Axis.FORWARD in caps.axes
             - yaw_rate: emitted in caps.yaw_unit (drone deg/s, rover rad/s)
             - down_m_s: emitted only when Axis.ALTITUDE in caps.axes

             Edge-zone behavior (ABS-05): when bbox is in the bottom/top
             margin (per config thresholds), emit forward_m_s=0. Fade-zone
             scaling + retreat push live in the adapter's _apply_retreat_from_tilt.

             Emergency safety (CONTEXT line 59): when bbox_height >
             max_bbox_height_safety AND Axis.FORWARD in caps.axes,
             emit forward_m_s=-max_backward + centering yaw.
             """
             # Yaw P (same math as today's compute_velocity_command yaw block)
             yaw_rate = _compute_yaw_p(detection, config)

             # Forward P + edge detection
             forward_m_s = 0.0
             if Axis.FORWARD in caps.axes:
                 forward_m_s = _compute_forward_p(detection, config)
                 # Bottom-edge detection: emit 0 if bbox bottom in margin
                 bbox_bottom = detection.center_y + detection.bbox_height / 2
                 bbox_top = detection.center_y - detection.bbox_height / 2
                 if config.bottom_margin_safety is not None and bbox_bottom > (1.0 - config.bottom_margin_safety):
                     forward_m_s = 0.0
                 if config.top_margin_safety is not None and bbox_top < config.top_margin_safety:
                     forward_m_s = 0.0
                 # Emergency safety (bbox too large)
                 if (config.max_bbox_height_safety is not None
                         and detection.bbox_height > config.max_bbox_height_safety):
                     forward_m_s = -config.max_backward
                     # Add centering-yaw recovery (same as today)
                     yaw_rate = _compute_centering_yaw(detection, config)

             # Down (nominal — adapter overlays altitude P correction)
             down_m_s = 0.0
             if Axis.ALTITUDE in caps.axes and not config.yaw_only:
                 # Today's controller didn't emit a non-trivial down — that's
                 # done by adapter's altitude P. Keep 0 here.
                 down_m_s = 0.0

             return RobotCommand(
                 forward_m_s=forward_m_s,
                 yaw_rate=yaw_rate,
                 down_m_s=down_m_s,
             )
         ```
